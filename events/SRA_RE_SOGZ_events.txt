			type = SRA_SOGZ_main_situation

			limit = {
				has_country_flag = has_SRA_Spear_Of_Galaxy_Zenith
				NOT = {
					any_owned_ship ={
						SRA_is_SOGZ_ship_size = yes
					}
				}
			}
			create_fleet = {
				name = "NAME_SRA_Spear_Of_Galaxy_Zenith"
				effect = {
					set_owner = event_target:SRA_Spear_Of_Galaxy_Zenith_own_country

						name = random
						design = "NAME_SRA_Bright_Galaxy_Adjust"
						upgradable = no
						effect = {
							set_disable_at_health = 0.01
						}
					}

						target = event_target:SRA_Spear_Of_Galaxy_Zenith_capital_location
						distance = 0
						angle = random
					}
					set_fleet_stance = aggressive
				}	
			}
		}
	}
	option = {
		name = SRA_RE_SOGZ.5.a
		hidden_effect = {
			country_event = { id = SRA_RE_SOGZ.6 }
			end_event_chain = SRA_RE_SOGZ_repair_chain
