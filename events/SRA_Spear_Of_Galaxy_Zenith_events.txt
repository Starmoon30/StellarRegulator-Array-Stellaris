			type = SRA_SOGZ_main_situation
		}

		set_variable = { which = SRA_diplomacy_bg value = 3 }
		set_country_flag = has_SRA_Spear_Of_Galaxy_Zenith
		set_country_flag = SRA_T4_weapon_planet_killer_killer_mode
		capital_scope = {
			save_event_target_as = SRA_Spear_Of_Galaxy_Zenith_capital_location
		}
		save_event_target_as = SRA_Spear_Of_Galaxy_Zenith_own_country
		if = {
			limit = {
				has_country_flag = has_SRA_Spear_Of_Galaxy_Zenith
				NOT = {
					any_owned_ship ={
						SRA_is_SOGZ_ship_size = yes
					}
				}
			}
			create_fleet = {
				name = "NAME_SRA_Spear_Of_Galaxy_Zenith"
				effect = {
					set_owner = event_target:SRA_Spear_Of_Galaxy_Zenith_own_country
					create_ship = {
						name = random
						design = "NAME_SRA_Bright_Galaxy"
						upgradable = no
						effect = {
							set_disable_at_health = 0.01
							save_event_target_as = SRA_Spear_Of_Galaxy_Zenith
							ship_event = {
								id = SRA_Spear_Of_Galaxy_Zenith.5023
							}
						}
					}			
					set_location = {
						target = event_target:SRA_Spear_Of_Galaxy_Zenith_capital_location
						distance = 0
						angle = random
					}
					set_fleet_stance = aggressive
