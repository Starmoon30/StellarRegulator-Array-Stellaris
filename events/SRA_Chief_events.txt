country_event = {
	id = SRA_Chief.1
	hide_window = yes

	is_triggered_only = yes

	fire_only_once = yes

	immediate = {
		set_update_modifiers_batch = begin
		set_global_flag = created_SRA_Chief
		create_country = {
			name = "NAME_SRA_Chief"
			type = "faction_SRA_Chief_country"
			flag = {
				icon = {
					category="SRA_flags"
					file="SRA_flags_Rokia_1.dds"
				}
				background={
					category="backgrounds"
					file="diagonal_stripe.dds"
				}
				colors={
					"blue"
					"black"
					"null"
					"null"
				}
			}
			name_list = SRA_Rocivia_Empire
			species = event_target:SRA_Chief_species
			effect = {
				set_country_flag = protected_from_queen_storm
				every_country = {
					prev = {
						set_closed_borders = {
							who = prev
							status = yes
						}
					}
				}
				SRA_enable_crisis_country_orbital_bombardment = yes
				save_global_event_target_as = SRA_Chief_country
				set_country_flag = StellarRegulator_Array_tech_permit
				set_country_flag = SRA_Chief_country
				add_resource = { 
					minerals = 50000
					energy = 50000	
					food = 50000
					alloys = 50000
					consumer_goods = 50000
					influence = 500 
				}
				set_graphical_culture = fallen_empire_01
				create_ship_design = { design = "NAME_SRA_Chief_Adjudicator" }
				add_ship_design = last_created_design
				create_ship_design = { design = "NAME_SRA_Chief_Executor" }
				add_ship_design = last_created_design
				create_ship_design = { design = "NAME_SRA_Chief_Cruiser_No_Aura" }
				add_ship_design = last_created_design
				create_ship_design = { design = "NAME_starbase_SRA_1" }
				add_ship_design = last_created_design
				create_ship_design = { design = "NAME_SRA_orbital_ring" }
				add_ship_design = last_created_design
				add_modifier = {
					modifier = SRA_Chief_Buff_0
				}
	
			}
		}
		if = {
			limit = { count_system = { count <= 250 }}
			country_event = { id = SRA_Chief.2 days = 1 }
		}
		else_if = {
			limit = { count_system = { count <= 450 } }
			country_event = { id = SRA_Chief.3 days = 1 }
		}
		else_if = {
			limit = { count_system = { count <= 650 } }
			country_event = { id = SRA_Chief.4 days = 1 }
		}
		else_if = {
			limit = { count_system = { count <= 850 } }
			country_event = { id = SRA_Chief.5 days = 1 }
		}
		else = {
			country_event = { id = SRA_Chief.6 days = 1 }
		}
		country_event = {
			id = SRA_Chief.7 days = 2
		}
		set_update_modifiers_batch = end
	}
}

country_event = {
	id = SRA_Chief.2
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		set_spawn_system_batch = begin
		no_scope = {
			# makes system positions originate from galactic core
			#set_spawn_system_batch = begin
			# batch-processes the spawn_system effects between "begin" and "end",
			# so caches are recalculated only once rather than for every system spawned
			spawn_system = {
				min_distance >= 200
				max_distance <= 210
				min_orientation_angle = 250
				max_orientation_angle = 260
				hyperlane = yes
				initializer = "SRA_Chief_colony_1_system"
				effect = {
					save_global_event_target_as = SRA_Chief_colony_1_system
				}
			}

				spawn_system = { 
					min_distance >= 20
					max_distance <= 50
					min_orientation_angle = 250
					max_orientation_angle = 260
					hyperlane = no
					initializer = "SRA_Chief_capital_system"
					effect = {
						save_global_event_target_as = SRA_Chief_capital_system
					}
				}
				spawn_system = { 
					min_distance >= 50
					max_distance <= 80

					initializer = "SRA_Chief_colony_2_system"

				}
