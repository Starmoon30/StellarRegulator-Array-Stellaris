	id = SRA_SD_event.10
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		is_planet_class = pc_SRA_SD_chaos_star_habitable
	}
	immediate = {
		random_list = {
			1000 = {
			}
			1000 = {
				multiply_variable = {
					which = pc_SRA_SD_chaos_star_habitable_vitality
					value = 1.1
				}
			}
			100 = {
				modifier = { 
					add = pc_SRA_SD_chaos_star_habitable_vitality
					always = yes
		    	}
				modifier = {
					mult = 0
					has_planet_flag = SRA_SD_abyss_no_army_attack
				}
				create_SRA_SD_abyss_army_attack_010 = yes
			}
		}

		if = {
			limit = {
				exists = owner
				owner = { is_gestalt = no }
				free_jobs_of_type = {
					job = SRA_SD_chaos_explorer
					value <= 0
				}
			}
			random_owned_pop_job = {
				limit = {
					has_job_type = SRA_SD_chaos_explorer
				}
				kill_assigned_pop_amount = {
					percentage = 1
				}
			}
			change_variable = {
				which = pc_SRA_SD_chaos_star_habitable_max_districts
				value = 1
			}
		}
		else_if = {
			limit = {
				exists = owner
				owner = { is_gestalt = yes }
				free_jobs_of_type = {
					job = SRA_SD_chaos_explorer_drone
					value <= 0
				}
			}
			random_owned_pop_job = {
				limit = {
					has_job_type = SRA_SD_chaos_explorer_drone
				}
				kill_assigned_pop_amount = {
					percentage = 1
				}
			}
			change_variable = {
				which = pc_SRA_SD_chaos_star_habitable_max_districts
				value = 1
			}
		}
		carrier_event = {
			id = SRA_SD_event.10 days = 180
		}
	}
}
