country_event = {
	id = SRA_Crisis.3
	hide_window = yes
	fire_only_once = yes
	is_triggered_only = yes
	trigger = {
		NOR = {
			has_global_flag = SRA_Crisis_disabled			#危机没有被禁用

			save_global_event_target_as = SRA_first_portal_system

				create_country = {
					name = "NAME_SRA_Crisis_country"
					type = "faction_SRA_Crisis_country"

                        icon= {
                            category="SRA_flags"
                            file="SRA_flags_Rocivia.dds"

                            file = "double_triangles.dds"
                        }

							"blue"
							"black"
                            "null"
                            "null"
                        }
                    }
					effect = {
						set_country_flag = protected_from_queen_storm
						set_country_flag = SRA_Crisis_country

						set_country_flag = community_can_orbital_bombard_ture

						save_global_event_target_as = SRA_Crisis_country
						set_global_flag = has_SRA_invasion_fleet_country
						set_global_flag = SRA_first_portal_system_opened

							limit = { has_global_flag = SRA_Crisis_normal }
							add_modifier = { modifier = SRA_Crisis_Buff_0 days = -1 }
						}
						else_if = {
							limit = { has_global_flag = SRA_Crisis_hard }
							add_modifier = { modifier = SRA_Crisis_Buff_1 days = -1 }
						}
						else_if = {
							limit = { has_global_flag = SRA_Crisis_lunatic }
							add_modifier = { modifier = SRA_Crisis_Buff_2 days = -1 }
						}
					}
				}
				create_SRA_Crisis_Guard_Fleet = yes
				create_SRA_Crisis_Guard_Fleet = yes
				event_target:SRA_Crisis_country = {
					create_fleet = {
						name = "NAME_SRA_Crisis_Fort_Fleet"
						effect = {
							set_owner = event_target:SRA_Crisis_country
							create_SRA_Crisis_Fort = yes
							set_location = {
								target = PREVPREV
								distance = 40
								angle = random
							}
							save_global_event_target_as = first_SRA_Crisis_Fort
							set_fleet_flag = first_SRA_Crisis_Fort
							set_event_locked = yes
						}
					}
					last_created_fleet = {
						set_name = "NAME_SRA_Crisis_Guard_Fleet"
						set_timed_fleet_flag = { flag = SRA_Crisis_Guard_fleet days = 6000 }
						fleet_event = { id = SRA_Crisis.401 days = 6001 }
					}
					set_variable = {
						which = SRA_Crisis_revive_1
						value = 0
					}
					set_variable = {
						which = SRA_Crisis_revive_2
						value = 0
					}
					set_variable = {
						which = SRA_Crisis_revive_3
						value = 0
					}
					set_variable = {
						which = SRA_Crisis_revive_4
						value = 0
					}
					set_variable = {
						which = SRA_Crisis_revive_5
						value = 0
					}
					set_variable = {
						which = SRA_Crisis_revive_6
						value = 0
					}
					#锚点debug
					country_event = { id = SRA_Crisis.1999 days = 30 }
					#狡猾同盟
					country_event = { id = SRA_Crisis.23366 }
					#突变
					country_event = { id = SRA_Mutation.4000 }
					# 4队进攻舰队
					country_event = { id = SRA_Crisis.3040  days = 50 }
					country_event = { id = SRA_Crisis.3040  days = 100 }
					country_event = { id = SRA_Crisis.3040  days = 200 }
					country_event = { id = SRA_Crisis.3040  days = 250 }
					if = {
						limit = { has_global_flag = SRA_Crisis_lunatic }
						# 额外4队进攻舰队
						country_event = { id = SRA_Crisis.3040  days = 100 }
						country_event = { id = SRA_Crisis.3040  days = 200 }
						country_event = { id = SRA_Crisis.3040  days = 400 }
						country_event = { id = SRA_Crisis.3040  days = 500 }
						#循环刷辅助守家舰队
					} else = { country_event = { id = SRA_Crisis.100 days = 5000 } }
					if = {
						limit = { has_global_flag = SRA_Crisis_normal }
						country_event = { id = SRA_Crisis.14 days = 750 }
					}
					else_if = {
						limit = { has_global_flag = SRA_Crisis_hard }
						country_event = { id = SRA_Crisis.14 days = 650 }
					} else = { country_event = { id = SRA_Crisis.14 days = 500 } }
				}
			}
		}
		every_playable_country = {
			if = {
				limit = {
					intel_level = {
						level = high
						system = event_target:SRA_first_portal_system
					}
				}
				country_event = { id = SRA_Crisis.5 }
			}
			if = {
				limit = {
					NOT = {
						intel_level = {
							level = high
							system = event_target:SRA_first_portal_system
						}
					}
				}
				country_event = { id = SRA_Crisis.6 }
			}
		}
		every_country = {
			limit = {
				is_country_type = faction_SRA_Crisis_country
			}
			country_event = { id = SRA_Crisis.15 days = 5 }
		}
		set_update_modifiers_batch = end
	}

carrier_event = {
	id = SRA_Crisis.10
	title = "SRA_Crisis.10.name"
	desc = "SRA_Crisis.10.desc"
	picture = GFX_evt_planet_beam
	show_sound = event_ghost_town
	is_triggered_only = yes
	trigger = {
		planet_devastation = 100
		OR = {
			FROM = { is_country_type = faction_SRA_Crisis_country }
			FROM = { is_country_type = faction_SRA_RE_Crisis_country_1 }
		}
	}
	immediate = {
		if = {
			limit = {
				has_global_flag = SRA_Mutation_5201
				any_fleet_in_orbit = {
					owner = {
						is_country_type = faction_SRA_Crisis_country
					}
					has_fleet_flag = SRA_attack_fleet
				}
		    }
			if = {
				limit = { has_global_flag = SRA_Crisis_normal }
				random_fleet_in_orbit = {
					limit = {
						owner = {
							is_country_type = faction_SRA_Crisis_country
						}
					}
					random = {
						chance = 50
						create_SRA_Crisis_fleet_and_admiral = yes
					}
				}
				event_target:SRA_Crisis_country = {
					if = {
						limit = { has_modifier = SRA_Mutation_5201 }
						change_variable = { which = SRA_Mutation_5201_variable value = 1 }
						add_modifier = { modifier = SRA_Mutation_5201 multiplier = SRA_Mutation_5201_variable days = 180 }
					}
					else = {
						set_variable = { which = SRA_Mutation_5201_variable value = 1 }
						add_modifier = { modifier = SRA_Mutation_5201 multiplier = SRA_Mutation_5201_variable days = 180 }
					}
				}
			}
			if = {
				limit = { has_global_flag = SRA_Crisis_hard }
				random_fleet_in_orbit = {
					limit = {
						owner = {
							is_country_type = faction_SRA_Crisis_country
						}
					}
					random = {
						chance = 50
						create_SRA_Crisis_fleet_and_admiral = yes
					}
					random = {
						chance = 50
						create_SRA_Crisis_fleet_and_admiral = yes
					}
				}
				event_target:SRA_Crisis_country = {
					if = {
						limit = { has_modifier = SRA_Mutation_5201 }
						change_variable = { which = SRA_Mutation_5201_variable value = 2 }
						add_modifier = { modifier = SRA_Mutation_5201 multiplier = SRA_Mutation_5201_variable days = 270 }
					}
					else = {
						set_variable = { which = SRA_Mutation_5201_variable value = 2 }
						add_modifier = { modifier = SRA_Mutation_5201 multiplier = SRA_Mutation_5201_variable days = 270 }
					}
				}
			}
			if = {
				limit = { has_global_flag = SRA_Crisis_lunatic }
				random_fleet_in_orbit = {
					limit = {
						owner = {
							is_country_type = faction_SRA_Crisis_country
						}
					}
					random = {
						chance = 50
						create_SRA_Crisis_fleet_and_admiral = yes
					}
					random = {
						chance = 50
						create_SRA_Crisis_fleet_and_admiral = yes
					}
					random = {
						chance = 50
						create_SRA_Crisis_fleet_and_admiral = yes
					}
					random = {
						chance = 50
						create_SRA_Crisis_fleet_and_admiral = yes
					}
				}
				event_target:SRA_Crisis_country = {
					if = {
						limit = { has_modifier = SRA_Mutation_5201 }
						change_variable = { which = SRA_Mutation_5201_variable value = 8 }
						add_modifier = { modifier = SRA_Mutation_5201 multiplier = SRA_Mutation_5201_variable days = 360 }
					}
					else = {
						set_variable = { which = SRA_Mutation_5201_variable value = 8 }
						add_modifier = { modifier = SRA_Mutation_5201 multiplier = SRA_Mutation_5201_variable days = 360 }
					}
				}
			}
		}
		# 相位星球
		if = {
			limit = { has_planet_flag = phaseshifting_active }
			remove_planet_flag = phaseshifting_active
			remove_modifier = phaseshifted
		}
		add_threat = { who = from amount = 2 }
		reset_planet = yes
		destroy_colony = yes
		if = {
			limit = { is_ringworld = yes }
			spawn_ringworld_cracker_effect = yes
			SRA_destroy_habitable_ringworld = yes
		}
		else_if = {
			limit = { SRA_is_habitat = yes }
			remove_planet = yes
		} else = { change_pc = pc_shattered }
		if = {
			limit = {
				FROM = { is_country_type = faction_SRA_Crisis_country }
			}
			every_country = {
				limit = { has_event_chain = "SRA_invasion_chain" }
				add_event_chain_counter = {
					event_chain = "SRA_invasion_chain"
					counter = "SRA_invastor_planets"
					amount = 1
				}
			}
		}
		if = {
			limit = {
				FROM = { is_country_type = faction_SRA_RE_Crisis_country_1 }
			}
			every_country = {
				limit = { has_event_chain = "SRA_RE_Crisis_chain_1" }
				add_event_chain_counter = {
					event_chain = "SRA_RE_Crisis_chain_1"
					counter = "SRA_RE_Crisis_1_planets"
					amount = 1
				}
			}
		}
	}
	option = { name = SRA_Crisis.10.a }
}

	hide_window = yes
	is_triggered_only = yes
	trigger = {
		NOT = {

		}
		OR = {
			AND = {
				has_global_flag = SRA_Crisis_lunatic
			    event_target:SRA_first_portal_system = {
				    count_ship_in_system = {
						limit = {
							owner = { is_country_type = faction_SRA_Crisis_country }
						} count < 400
					}
				}
			}
			event_target:SRA_first_portal_system = {
				count_ship_in_system = {
					limit = {
						owner = { is_country_type = faction_SRA_Crisis_country }
					} count < 200
				}
			}
		}
	}
	immediate = {
		if = {
			limit = { check_variable = { which = SRA_Crisis_revive_1 value < 10 } }
			event_target:SRA_first_portal_system = {
				random_system_planet = {
					create_SRA_Crisis_Guard_Fleet = yes
					last_created_fleet = {
					set_name = "NAME_SRA_Crisis_Guard_Fleet"
					set_timed_fleet_flag = { flag = SRA_defence_fleet days = 4500 }
					fleet_event = { id = SRA_Crisis.401 days = 4501 }
				    }
			    }
		    }
			change_variable = {
				which = SRA_Crisis_revive_1
				value = 1
			}
		    country_event = {
		    	id = SRA_Crisis.2201 days = 2
		    }
		}
		else = {
			set_variable = {
				which = SRA_Crisis_revive_1
				value = 0
			}

		    	id = SRA_Crisis.2201 days = 2
		    }

					limit = {
						owner = {

							}
