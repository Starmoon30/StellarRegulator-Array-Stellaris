ship_event = {
	id = SRA_Barrier.100
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ship_size = SRA_Barrier_Main_Station
		exists = owner
	}
	immediate = {
		fleet = {
			set_event_locked = yes
			fleet_event = {
				id = SRA_Barrier.101 days = 180
			}
		}
	}
}
fleet_event = {
	id = SRA_Barrier.101
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ship_size = SRA_Barrier_Main_Station
		exists = owner
	}
	immediate = {
		if = {
			limit = {
				solar_system = {
					any_system_planet = {
						is_colony = yes
					}
				}
			}
			fleet_event = {
				id = SRA_Barrier.102
			}
		}
		else = {
			fleet_event = {
				id = SRA_Barrier.103
			}
		}
	}
}

fleet_event = {
	id = SRA_Barrier.102
	hide_window = yes

				}

					id = SRA_Barrier.1
				}
			}
		}
		delete_fleet = this
	}
}

carrier_event = {
	id = SRA_Barrier.111
	hide_window = yes
	is_triggered_only = yes
	pre_triggers = {
		has_ground_combat = yes
	}
	trigger = {
		colony = {
			check_modifier_value = {
				modifier = mod_SRA_Planet_Combat_Width_Attacker
				value >= 1
			}
		}
	}
	immediate = {
		if	= {
			limit = {
				colony = {
					check_modifier_value = {
						modifier = mod_SRA_Planet_Combat_Width_Attacker
						value < prev.trigger:count_ground_combat_attacker
					}
				}
			}
			set_variable = {
				which = SRA_Planet_Combat_Width_Attacker
				value = colony.modifier:mod_SRA_Planet_Combat_Width_Attacker
			}
			multiply_variable = {
				which = SRA_Planet_Combat_Width_Attacker
				value = -1
			}
			change_variable = {
				which = SRA_Planet_Combat_Width_Attacker
				value = trigger:count_ground_combat_attacker
			}
			while = {
				count = SRA_Planet_Combat_Width_Attacker
				random_ground_combat_attacker = {
					owner = {
						fire_on_action = {
							on_action = on_army_killed_in_combat
							scopes = {
								From = prev
								FromFrom = prevprev.owner
								FromFromFrom = prevprev
							}
						}
					}
					remove_army = yes
				}
			}
			clear_variable = SRA_Planet_Combat_Width_Attacker
		}
		carrier_event = {
			id = SRA_Barrier.111 days = 1
		}
	}
}
