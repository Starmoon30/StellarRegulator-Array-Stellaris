								on_action = on_ship_destroyed_perp

									from = prev.owner
									fromfrom = event_target:SRA_target_ship
									fromfromfrom = prev
								}
							}

						fire_on_action = {
							on_action = on_fleet_destroyed_victim
							scopes = {
								from = root.from.owner
								fromfrom = prev
								fromfromfrom = root.from
							}
						}
					}

						fire_on_action = {
							on_action = on_fleet_destroyed_perp
							scopes = {
								from = prev.owner
								fromfrom = root.from
								fromfromfrom = prev
							}
						}
					}
				}
			}
			every_fleet_in_system = {
				limit = {
					exists = this
					NOR = {
						is_same_value = from
						any_owned_ship = {
							is_ship_class = shipclass_starbase

								is_ship_size = ag_gamma_station_small_hostile
								is_ship_size = ag_gamma_station_small_moving_hostile
								is_ship_size = juggernaut
							}
						}
					}
				}
				every_controlled_ship = {
					if = {
						limit = {
							is_disabled = no
						}
						SRA_clear_ship_damage_reduction_mult = {
							ALSO_LESS_THAN_ZERO = yes
						}
						damage_ship = {
							amount = 80000000000000
							attacker = event_target:SRA_target_ship
						}
					}
				}
			}
		}

			create_message = {
				type = "STAR_DESTROYED"

				variable = {
					type = name

				}
				variable = {
					type = name
					localization = "SRA_target_system"
					scope = root.solar_system
				}
				variable = {
					type = name
					localization = "SRA_attacker"
					scope = from.owner
				}
				variable = {
					type = name
					localization = "SRA_attacker_ship"
					scope = from
				}
			}
		}
		set_update_modifiers_batch = end
	}
