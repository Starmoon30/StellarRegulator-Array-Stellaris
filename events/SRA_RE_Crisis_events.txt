country_event = {
	id = SRA_RE_Crisis.10
	hide_window = yes
	is_triggered_only = yes
	trigger = { is_country_type = faction_SRA_RE_Crisis_country_1 }
	immediate = {
		random_system = {
			limit = { has_star_flag = SRA_RE_Crisis_1_system }
			if = {
				limit = {
					NOT = { exists = starbase }
				}
				create_starbase = {
					size = starbase_SRA_RE_Calamitisease
					owner = ROOT
					#module = <starbase_module>
					#building = <starbase_building>
					effect = {
						set_ship_construction_type = starbase_shipyard
					}
				}
			}
			random_system_planet = {
				create_SRA_RE_Crisis_1_fleet = yes
			}
			every_fleet_in_system = {
				if = {
					limit = {
						check_variable = {
							which = SRA_RE_Crisis_1_system_variable
							value < solar_system.SRA_RE_Crisis_1_system_variable 
						}
					}
					set_variable = {
						which = SRA_RE_Crisis_1_system_variable 
						value = solar_system.SRA_RE_Crisis_1_system_variable 
					}
				}
				if = {
					limit = { owner = { is_same_value = Root } }
					add_modifier = { modifier = SRA_RE_Crisis_Buff_10 multiplier = SRA_RE_Crisis_1_system_variable days = -1 }
				}
				else = {
					add_modifier = { modifier = SRA_RE_Crisis_Buff_11 multiplier = SRA_RE_Crisis_1_system_variable days = 1800 }
				}
			}
		}
		country_event = { id = SRA_RE_Crisis.10 days = 360 }
	}

country_event = {
	id = SRA_RE_Crisis.20
	hide_window = yes
	is_triggered_only = yes
	trigger = { is_country_type = faction_SRA_RE_Crisis_country_1 }
	immediate = {
		if = {
			limit = {
				FROM = { has_event_chain = "SRA_RE_Crisis_chain_1" }
			}
			FROM = {
				add_event_chain_counter = {
					event_chain = "SRA_RE_Crisis_chain_1"
					counter = "SRA_RE_Crisis_1_kills_us"
					amount = 1
				}
				if = {
					limit = { has_civic = civic_SRARociviaEmpire }
					change_variable = {
						which = SRA_RE_Crisis_1_stage_counter
						value = 1
					}
				}
				if = {
					limit = {
					    check_variable = { which = SRA_RE_Crisis_1_stage_counter value >= 1000 }
					    NOT = { has_global_flag = SRA_RE_Crisis_1_stage_2 }
				    }
					if = {
						limit = { NOT = { has_global_flag = SRA_no_crisis_sound } }
						stop_crisis_sound = yes
						set_crisis_sound = BGM_CalamitiseaseStage2
					}
					set_global_flag = SRA_RE_Crisis_1_stage_2
				}
				if = {
					limit = {
					    check_variable = { which = SRA_RE_Crisis_1_stage_counter value >= 2000 }
					    NOT = { has_global_flag = SRA_RE_Crisis_1_stage_3 }
				    }
					if = {
						limit = { NOT = { has_global_flag = SRA_no_crisis_sound } }
						stop_crisis_sound = yes
						set_crisis_sound = BGM_CalamitiseaseStage3
					}
					set_global_flag = SRA_RE_Crisis_1_stage_3
				}
				if = {
					limit = {
					    check_variable = { which = SRA_RE_Crisis_1_stage_counter value >= 3000 }
					    NOT = { has_global_flag = SRA_RE_Crisis_1_stage_4 }
				    }
					if = {
						limit = { NOT = { has_global_flag = SRA_no_crisis_sound } }
						stop_crisis_sound = yes
						set_crisis_sound = BGM_CalamitiseaseStage4
					}
					set_global_flag = SRA_RE_Crisis_1_stage_4
				}
				if = {
					limit = {
					    check_variable = { which = SRA_RE_Crisis_1_stage_counter value >= 4000 }
					    NOT = { has_global_flag = SRA_RE_Crisis_1_stage_5 }
				    }
					if = {
						limit = { NOT = { has_global_flag = SRA_no_crisis_sound } }
						stop_crisis_sound = yes
						set_crisis_sound = BGM_CalamitiseaseStage5
					}
					set_global_flag = SRA_RE_Crisis_1_stage_5
					country_event = {
						id = SRA_RE_Crisis.40
					}
				}
			}
		} else = {
			every_country = {
				limit = {
					has_event_chain = "SRA_RE_Crisis_chain_1"
					NOT = { is_same_value = FROM }
				}
				add_event_chain_counter = {
					event_chain = "SRA_RE_Crisis_chain_1"
					counter = "SRA_RE_Crisis_1_kills_others"
					amount = 1
				}
			}
		}

country_event = {
	id = SRA_RE_Crisis.142
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_ai = no
		has_civic = civic_SRARociviaEmpire
	}
	immediate = {
		if = {
			#六次后未选完，失败
			limit = {
				has_country_flag = SRA_RE_Crisis_1_End_Chosen_6
				NOT = {
					has_country_flag = SRA_RE_StellarRegulator_debris_9
				}
			}
			country_event = {
				id = SRA_RE_Crisis.143
			}
		}
		if = {
			#按顺序选择
			limit = {
