namespace = SRA_Megastructures

#星环基地被摧毁
ship_event = {
	id = SRA_Megastructures.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ship_size = SRA_orbital_ring
		exists = owner
	}
	immediate = {
		set_disable_at_health = 1.01
		repair_ship = yes
		fleet = {
			fleet_event = {
				id = SRA_Megastructures.2 days = 180
			}
		}
	}
}
fleet_event = {
	id = SRA_Megastructures.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ship_size = SRA_orbital_ring
		exists = owner
	}
	immediate = {
		if = {
			limit = {
				orbit = {
					is_colony = yes
				}
			}
			set_owner = orbit.controller
			every_controlled_ship = {
				repair_ship = yes
				set_disable_at_health = 0.01
				set_disabled = no
			}
		}
		else = {
			destroy_fleet = {
				target = this
				kill_leader = no
				destroy_template = yes
			}
		}
	}
}



# This = Ship
# From = Disabler Ship
# Habitat Orbital Disabled
ship_event = {
	id = SRA_Megastructures.115
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		is_ship_class = shipclass_habitat_station
		FROM.owner = { SRA_can_do_orbital_bombardment_country = yes }
	}

	immediate = {
		if = {
			limit = {
				exists = orbit
			}
			orbit = {
				remove_planet_flag = has_megastructure
				save_event_target_as = target_planet
			}
		}
		solar_system = {
			random_system_planet = {
				limit = {
					has_planet_flag = habitat
				}
				every_deposit = {
					limit = {
						has_deposit_flag = planet@event_target:target_planet
					}
					remove_deposit = yes
				}
			}
		}
		destroy_ship = this
		create_ambient_object = {
			location = this
			type = habitat_cracker_object
			duration = 5

			use_3d_location = yes
			base_angle_towards = star
			entity_face_object = star

			entity_offset = { min = 0 max = 0 }

			entity_scale_to_size = yes
			scale = 0.10
		}
	}
}
