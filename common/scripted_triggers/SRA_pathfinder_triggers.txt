SRA_pathfinder_standard_system_target_triggers = {
	optimize_memory
	OR = {
		AND = {
			exists = controller
			controller = { OR = { is_hostile = event_target:$OWNER$ is_hostile = event_target:$OWNER_2$ } }
			OR = { is_colony = yes is_under_colonization = yes }
		}
		AND = {
			always = $ATTACK_COLONIZABLE$
			is_colonizable = yes
		}
	}
	count_fleet_in_orbit = { count = 0 limit = {
		exists = owner
		owner = { OR = { is_same_value = event_target:$OWNER$ is_same_value = event_target:$OWNER_2$ } }
		NOT = { is_same_value = root }
	} }
}

SRA_pathfinder_standard_ship_target_triggers = {
	optimize_memory
	exists = owner
	owner = { OR = { is_hostile = event_target:$OWNER$ is_hostile = event_target:$OWNER_2$ } }
	is_disabled = no 
	OR = { is_ship_class = shipclass_military is_ship_class = shipclass_military_special is_ship_class = shipclass_starbase }
}
