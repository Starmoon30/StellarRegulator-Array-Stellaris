councilor_SRA_RE_minister_of_security = {
	leader_class = { official commander scientist }
	is_leader_possible = {
		OR = {
			has_trait = leader_trait_SRA_RE_Hidra_gov
			has_trait = leader_trait_SRA_RE_Hidra_sci
			has_trait = leader_trait_SRA_RE_Hidra_adm
			has_trait = leader_trait_SRA_RE_Hidra_gen
		}
	}
	possible = {
		has_civic = civic_SRARociviaEmpire
		has_country_flag = SRA_RE_Rocivia_Empire_story
	}
	icon = "GFX_SRARociviaEmpire"
	modifier = {
		planet_stability_add = 1
		all_technology_research_speed = 0.05
		planet_crime_add = -5
		planet_army_build_speed_mult = 0.1
	}
}

councilor_SRA_SD_lost_presbyter = {
	leader_class = { official commander scientist }

			leader_class = commander
			leader_class = scientist
		}
		not = { has_trait = leader_trait_SRA_SD_Bard_of_Metarbor }
	}
	possible = {
		has_country_flag = SRA_SD_Shadow_Of_Hidra_story
	}
	icon = "GFX_SRA_RE_Hidra_icon"
	modifier = {
		planet_jobs_produces_mult = 0.1
	}
}
