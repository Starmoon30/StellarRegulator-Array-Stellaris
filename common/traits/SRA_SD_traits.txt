leader_trait_SRA_SD_Myth = {
	inline_script = {
		script = trait/SRA_common_trait_template
		SRA_council = no
		SRA_rarity = paragon		# common, veteran, paragon, free_or_veteran
		SRA_class = leader			# official, scientist, commander
		SRA_background = GFX_SRA_r_trait_glow_destiny
		SRA_icon = GFX_leader_trait_SRA_SD_Myth
		SRA_tier = none
	}
	triggered_modifier = {
		potential = { always = yes }
		custom_tooltip = SRA_SD_no_custom_tooltip
		mod_SRA_SD_Myth_power = 1
		mult = this.SRA_SD_Myth_power_level
	}
	triggered_desc = {
		trigger = { always = yes }
		text = SRA_SD_Myth_power_level
	}
	triggered_self_modifier = {
		potential = { has_leader_flag = SRA_SD_Myth_councilor_skill }
		leader_skill_add = 1
		mult = this.SRA_SD_Myth_power_level
	}
	triggered_desc = {
		trigger = { has_leader_flag = SRA_SD_Myth_councilor_skill }
		text = SRA_SD_Myth_councilor_skill
	}
	triggered_fleet_modifier = {
		potential = { has_leader_flag = SRA_SD_Myth_damage_mult }
		ship_shield_damage_mult = 0.1
		ship_armor_damage_mult = 0.1
		ship_hull_damage_mult = 0.1
		mult = this.SRA_SD_Myth_power_level
	}
	triggered_desc = {
		trigger = { has_leader_flag = SRA_SD_Myth_damage_mult }
		text = SRA_SD_Myth_damage_mult
	}
	triggered_fleet_modifier = {
		potential = { has_leader_flag = SRA_SD_Myth_evasion_mult }
		ship_evasion_mult = 0.1
		ship_accuracy_mult = 0.1
		ship_tracking_mult = 0.1
		mult = this.SRA_SD_Myth_power_level
	}
	triggered_desc = {
		trigger = { has_leader_flag = SRA_SD_Myth_evasion_mult }
		text = SRA_SD_Myth_evasion_mult
	}
	triggered_self_modifier = {
		potential = { has_leader_flag = SRA_SD_Myth_accuracy_debuff }
		custom_tooltip = SRA_SD_no_custom_tooltip
		mod_SRA_SD_Myth_accuracy_debuff = 1
		mult = this.SRA_SD_Myth_power_level
	}
	triggered_desc = {
		trigger = { has_leader_flag = SRA_SD_Myth_accuracy_debuff }
		text = SRA_SD_Myth_accuracy_debuff
	}
	triggered_army_modifier = {
		potential = {
			has_leader_flag = SRA_SD_Myth_army_damage_taken
		}
		army_damage_taken_mult = -0.1
		mult = this.SRA_SD_Myth_power_level
	}
	triggered_desc = {
		trigger = { has_leader_flag = SRA_SD_Myth_army_damage_taken }
		text = SRA_SD_Myth_army_damage_taken
	}
	triggered_fleet_modifier = {
		potential = {
			fleet = { num_ships <= 1 }
			has_leader_flag = SRA_SD_Myth_alone
		}
		ship_shield_damage_mult = 0.1
		ship_armor_damage_mult = 0.1
		ship_hull_damage_mult = 0.1
		ship_evasion_mult = 0.1
		ship_accuracy_mult = 0.1
		ship_tracking_mult = 0.1
		ship_shield_regen_add_static = 256000
		ship_armor_regen_add_static = 256000
		ship_hull_regen_add_static = 256000
		mult = this.SRA_SD_Myth_power_level
	}
	triggered_desc = {
		trigger = { has_leader_flag = SRA_SD_Myth_alone }
		text = SRA_SD_Myth_alone
	}
	triggered_sector_modifier = {
		potential = {
			has_leader_flag = SRA_SD_Myth_sector_speed
		}
		ship_speed_mult = 1
		mult = this.SRA_SD_Myth_power_level
	}
	triggered_desc = {
		trigger = { has_leader_flag = SRA_SD_Myth_sector_speed }
		text = SRA_SD_Myth_sector_speed
	}
	on_gained_effect = {
		if = {
			limit = { has_trait = leader_trait_SRA_Abyss_Erosion_Alter }
			remove_trait = leader_trait_SRA_Abyss_Erosion_Alter
			remove_all_negative_traits = yes
			set_leader_flag = leader_death_events_blocked
			set_leader_tier = leader_tier_legendary
			set_leader_flag = immune_to_negative_traits
			set_variable = { which = SRA_SD_Myth_power_level value = 1 }
		}
		else = {
			remove_trait = leader_trait_SRA_SD_Myth
		}
	}
	leader_class = { official commander scientist }
	initial = no
	randomized = no
}
