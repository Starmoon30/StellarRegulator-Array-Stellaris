SRA_SOGZ_menu_sub_1_1 = {
	potential = { always = yes }
	allow = { always = yes }
	effect = {
		FROM = {
			custom_tooltip = SRA_SOGZ_menu_sub_1_1_custom_tooltip
			hidden_effect = {
				if = {
					limit = {
						NOT = { has_country_flag = SRA_SOGZ_menu_sub_unlimited }
					}
					remove_country_flag = SRA_SOGZ_menu_sub_1_2_choose
					remove_country_flag = SRA_SOGZ_menu_sub_1_3_choose
				}
				set_country_flag = SRA_SOGZ_menu_sub_1_1_choose
				SRA_SOGZ_sub_energy_choose_check_effect = yes
			}
		}
	}
}
