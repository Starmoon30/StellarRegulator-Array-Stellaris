SRA_BoM_stat_a_effect = {
    potential = {}
    allow = {}
    effect = {
		custom_tooltip = "SRA_BoM_stat_a_tooltip"
        hidden_effect = {
            if = {
                limit = {
                    FROM = { check_variable = { which = SRA_BoM_stat_choose value = 1 } }
                }
                FROM = { set_variable = { which = SRA_BoM_stat_choose value = 0 } }
            }
            else = {
                FROM = { set_variable = { which = SRA_BoM_stat_choose value = 1 } }
            }
        }
    }
}
SRA_BoM_stat_a_box_effect = {
	potential = { hidden_trigger = { FROM = { check_variable = { which = SRA_BoM_stat_choose value = 1 } } } }
	allow = { always = yes }
    effect = { custom_tooltip = "SRA_settings_box" }
}
