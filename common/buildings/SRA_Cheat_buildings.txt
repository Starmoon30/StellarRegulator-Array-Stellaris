building_SRA_databank = {
	base_buildtime = 100
	can_build = yes
	can_demolish = yes
	icon = building_SRA_databank
	category = research
	position_priority = 3
	destroy_trigger = { always = no }
	potential = {
	owner? = {
		OR = {
			custom_tooltip = {
				fail_text = "SRA_need_building_core"
				check_variable = {
					which = building_SRA_databank_counter
					value > 0
				}
			}
			AND = {
				has_country_flag = StellarRegulator_Array_tech_permit
				OR = {
					has_civic = civic_SRAtechnology
					has_civic = civic_SRARociviaEmpire
					is_country_type = faction_SRA_Chief_country
				}
			}
		}
	}
	}
	
	inline_script = "buildings/SRA_building_sets_all"
	on_built = {
		owner = {
			if = {
				limit = {
					check_variable = {
						which = building_SRA_databank
						value > 0
					}
				}
				subtract_variable = {
					which = building_SRA_databank
					value = 1
				}
			}
			else = {
				set_variable = {
					which = building_SRA_databank
					value = 0
				}
			}
		}
	}
	resources = {
		category = planet_buildings
		cost = {
			minerals = 200
		}
		produces = {
			physics_research = 6000
			society_research = 6000
			engineering_research = 6000
			unity = 4000
		}
		upkeep = {
		}
	}
}
