			}

	}

SRA_nano_shipyard = {
	icon = "GFX_SRA_nano_shipyard"
	construction_days = 120
	category = sm_naval_support_cat
	resources = {
		category = starbase_buildings
		cost = { alloys = 1600 }
		upkeep = { energy = 16 alloys = 8 }
	}
	triggered_station_modifier = {
		potential = { fleet = { num_ships > 1 } }
		ship_target_weight_mult = -1
	}
	station_modifier = {
		starbase_shipyard_capacity_add = 480
		starbase_shipyard_build_speed_mult = 1
	}
	ship_modifier = {
		ship_starting_experience_add = 10000
	}

	potential = {
		exists = owner
		owner = { has_technology = "tech_SRA_nano_shipyard" }
		custom_tooltip = {
			fail_text = "requires_starport"
			OR = {
				has_starbase_size >= starbase_starport
				has_starbase_size >= SRA_orbital_ring
			}
		}
	}	
	possible = {
		custom_tooltip = {
			fail_text = "requires_starport"
			OR = {
				has_starbase_size >= starbase_starport
				has_starbase_size >= SRA_orbital_ring
			}
		}
		NOT = { has_starbase_building = SRA_T3_shipyard }
	}	
	show_in_tech = "tech_SRA_nano_shipyard"
	ai_build_at_chokepoint = no
	ai_build_outside_chokepoint = yes
	on_finished = {
		set_ship_construction_type = starbase_shipyard
	}
	ai_weight = {
		weight = 0
		modifier = {
			weight = 5000
			from = {
				OR = {
					has_civic = civic_SRAtechnology
					has_civic = civic_SRARociviaEmpire
				}
			}
		}
	}
}
