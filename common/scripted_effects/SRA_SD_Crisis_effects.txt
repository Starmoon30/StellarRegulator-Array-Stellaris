SRA_SD_Crisis_Pathfinder_Initialize = {
	optimize_memory
    # TARGET参数控制舰船的终点星系，建立寻路系统的时候会为每个星系附上带有终点星系标签的变量以判断每个星系与终点星系的超空间距离
    every_system = {
        export_trigger_value_to_variable = {
            trigger = distance
            parameters = {
                source = root
                type = hyperlane
                use_bypasses = yes
            }
            variable = SRA_SD_Crisis_Pathfinder_value_$TARGET$
        }
    }
}
SRA_SD_Crisis_Pathfinder = {
	optimize_memory
    # TARGET参数控制舰船的终点星系，建立寻路系统的时候会为每个星系附上带有终点星系标签的变量以判断每个星系与终点星系的超空间距离
    set_variable = {
        which = SRA_SD_Crisis_Pathfinder_value
        value = solar_system.SRA_SD_Crisis_Pathfinder_value_$TARGET$
    }
    if = {
        limit = {
            check_variable = {
                which = SRA_SD_Crisis_Pathfinder_value
                value > 0
            }
        }
        ordered_system = {
            limit = {
                check_variable = {
                    which = SRA_SD_Crisis_Pathfinder_value_$TARGET$
                    value < prev.SRA_SD_Crisis_Pathfinder_value
                }
            }
            position = 0
            order_by = trigger:distance
            if = {
                limit = { prev = { can_access_system = prev } }
                if = {
                    limit = { any_system_planet = { OR = { is_colony = yes is_under_colonization = yes } } }
                    random_system_planet = {
                        limit = {  OR = { is_colony = yes is_under_colonization = yes }  }
                        prevprev = {
                            auto_move_to_planet = {
                                target = PREV
                                clear_auto_move_on_arrival = yes
                            }
                        }
                    }
                }
                else = {
                    random_system_planet = {
                        prevprev = {
                            auto_move_to_planet = {
                                target = PREV
                                clear_auto_move_on_arrival = yes
                            }
                        }
                    }
                }
            }
            else = {
                random_system_planet = {
                    prevprev = {
                        set_location = { target = PREV distance = 5 angle = random }
                    }
                }
            }
        }
    }
}
