create_StellarRegulator_ruler = {
	optimize_memory
	create_leader = {
		class = official
		species = event_target:SRA_StellarRegulator_species
		name = "Name_SRA_leader_StellarRegulator"
		event_leader = yes
		gender = female
		immortal = yes
		skill = 10
		set_age = 14
		skip_background_generation = yes
		custom_catch_phrase = SRA_StellarRegulator_catch_phrase
		custom_description = SRA_StellarRegulator_backstory
		background_ethic = ethic_materialist
		hide_age = yes
		traits = {
			trait = subclass_official_economy_councilor
			trait = leader_trait_SRA_ruler
			trait = trait_ruler_charismatic_2
			trait = leader_trait_psionic
		}
		effect = {
			set_leader_flag = leader_death_events_blocked
			set_leader_tier = leader_tier_legendary
			set_leader_flag = immune_to_negative_traits
			change_background_job = SRA_StellarRegulator_job
			change_background_ethic = ethic_gestalt_consciousness
			inline_script = {
				script = scripted_effects/SRA_set_StellarRegulator_leader_portrait
			}
			owner = {
				set_leader = prev
			}
			if = {
				limit = { owner = { is_gestalt = yes } }
				remove_subclass = yes
				add_trait = { trait = subclass_official_economy_councilor show_message = no }
				add_trait = { trait = leader_trait_SRA_ruler show_message = no }
			}
		}
	}
}

create_SRA_RE_Rocivia_ruler = {
	optimize_memory
	create_leader = {
		class = official
		species = event_target:SRA_RE_species
		name = "Name_SRA_RE_leader_Rocivia"
		event_leader = yes
		gender = female
		immortal = yes
		skill = 10
		set_age = 16
		skip_background_generation = yes
		custom_catch_phrase = SRA_RE_Rocivia_catch_phrase
		custom_description = SRA_RE_Rocivia_backstory
		background_ethic = ethic_materialist
		hide_age = yes
		traits = {
			trait = subclass_official_economy_councilor
			trait = leader_trait_SRA_RE_ruler1
			trait = leader_trait_SRA_RE_ruler2
			trait = leader_trait_psionic
			trait = leader_trait_legendary_natural_ruler
		}
		effect = {
			set_leader_flag = restored_node
			set_leader_flag = leader_death_events_blocked
			set_leader_tier = leader_tier_legendary
			set_leader_flag = immune_to_negative_traits
			change_background_job = SRA_RE_Rocivia_job
			change_leader_portrait = SRA_portrait
			set_variable = { which = SRA_portrait_clothes_var value = 202 }
			if = {
				limit = {
					owner.ruler = {
						has_trait = leader_trait_legendary_leader
					}
				}
				owner.ruler = {
					remove_trait = leader_trait_legendary_leader
					remove_leader_flag = paragon_origin_legendary_leader
				}
				set_leader_flag = paragon_origin_legendary_leader
				add_trait = { trait = leader_trait_legendary_leader show_message = no }
			}
			owner = {
				set_leader = prev
				every_owned_leader = {
					limit = {
						has_trait = leader_trait_SRA_RE_ruler1
						NOT = { is_same_value = ruler }
					}
					kill_leader = { show_notification = no fire = yes }
				}
			}
			if = {
				limit = { owner = { is_gestalt = yes } }
				remove_subclass = yes
				add_trait = { trait = subclass_official_economy_councilor show_message = no }
				add_trait = { trait = leader_trait_SRA_RE_ruler1 show_message = no }
				add_trait = { trait = leader_trait_SRA_RE_ruler2 show_message = no }
				add_trait = { trait = leader_trait_legendary_natural_ruler show_message = no }
			}
		}
	}
}
