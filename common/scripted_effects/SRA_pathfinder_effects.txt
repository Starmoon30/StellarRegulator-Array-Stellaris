SRA_pathfinder_arrival_planet_action = {
	# this = fleet
	# from = planet
	# remove_auto_move_target = yes
	from = {
		if = {
			#未殖民宜居星球过1个月炸掉
			limit = { is_colonizable = yes }
			carrier_event = { id = SRA_Crisis.47 days = 30 scopes = { from = root } }
		}
		if = {
			#已殖民宜居星球过3个月炸掉
			limit = { is_colony = yes }
			carrier_event = { id = SRA_Crisis.47 days = 90 scopes = { from = root } }
		}
	}
}
SRA_pathfinder_jump_drive_planet_action  = {
	random_list = {
		95 = {
			prevprev = { 
				auto_move_to_planet = { target = prev clear_auto_move_on_arrival = no arrival_effect = SRA_pathfinder_arrival_planet_action }
			}
		}
		5 = {
			prevprev = {
				set_location = { target = prev distance = 30 angle = random }
				every_owned_ship = { 
					create_ambient_object = {
						type = horror_spawn_object
						location = this
						use_3d_location = yes
						entity_offset = { min = 0 max = 0 }
						entity_offset_angle = { min = 0 max = 0 }
						entity_offset_height = { min = 0 max = 0 }
						entity_scale_to_size = yes
						scale = @SRA_pathfinder_jumpdrive_effect_scale
					}
				}
				auto_move_to_planet = { target = prev clear_auto_move_on_arrival = no arrival_effect = SRA_pathfinder_arrival_planet_action }
			}
		}
	}
}

SRA_pathfinder_fleet_wandering_action = {
	optimize_memory
	clear_orders = yes
	clear_fleet_actions = this
	random_list = {
		#周围星系随机设置目标
		95 = {
			solar_system = { random_neighbor_system = {
				limit = { prevprev = { can_access_system = prev } }
				save_event_target_as = $TYPE$_invasion_fleet_wandering_action_target
			} }
			if = {
				limit = { exists = event_target:$TYPE$_invasion_fleet_wandering_action_target }
				event_target:$TYPE$_invasion_fleet_wandering_action_target = {
					system_star = { prevprev = { auto_move_to_planet = { target = prev clear_auto_move_on_arrival = yes } } }
				}
			} 
			else_if = {
				limit = { any_system = {
					prev = { can_access_system = prev }
				} }
				random_system = {
					limit = { prev = { can_access_system = prev } }
					save_event_target_as = $TYPE$_invasion_fleet_wandering_action_target
				}
				event_target:$TYPE$_invasion_fleet_wandering_action_target = {
					system_star = { prevprev = { auto_move_to_planet = { target = prev clear_auto_move_on_arrival = yes } } }
				}
			}
			else = { 
				log = "跃迁强度：[$TYPE$_fleet_none_target_var]"
				change_variable = { which = $TYPE$_fleet_none_target_var value = 1 }
			}
		}
		#找任意一个敌对目标，折跃
		5 = {
			modifier = {
				mult = $TYPE$_fleet_none_target_var
				is_variable_set = $TYPE$_fleet_none_target_var
			}
			clear_variable = $TYPE$_fleet_none_target_var
			random_system = {
				limit = {
					NOT = { is_same_value = prev.solar_system }
					OR = {
						any_system_planet = { SRA_pathfinder_standard_system_target_triggers = { OWNER = $OWNER$ OWNER_2 = $OWNER_2$ ATTACK_COLONIZABLE = $ATTACK_COLONIZABLE$ } }
						any_ship_in_system = { SRA_pathfinder_standard_ship_target_triggers = { OWNER = $OWNER$ OWNER_2 = $OWNER_2$ } }
					}
				}
				save_event_target_as = $TYPE$_invasion_fleet_wandering_action_target
			}
			if = {
				limit = { exists = event_target:$TYPE$_invasion_fleet_wandering_action_target }
				event_target:$TYPE$_invasion_fleet_wandering_action_target = {
					random_system_planet = {
						limit = { is_primary_star = yes }
						prevprev = {
							set_location = { target = prev distance = 300 angle = random }
							every_owned_ship = { 
								create_ambient_object = {
									type = horror_spawn_object
									location = this
									use_3d_location = yes
									entity_offset = { min = 0 max = 0 }
									entity_offset_angle = { min = 0 max = 0 }
									entity_offset_height = { min = 0 max = 0 }
									entity_scale_to_size = yes
									scale = @SRA_pathfinder_jumpdrive_effect_scale
								}
							}
							auto_move_to_planet = { target = prev clear_auto_move_on_arrival = yes }
						}
					}
				}
			} 
			else = {
				solar_system = { random_neighbor_system = {
					limit = { prevprev = { can_access_system = prev } }
					save_event_target_as = $TYPE$_invasion_fleet_wandering_action_target
				} }
				if = {
					limit = { exists = event_target:$TYPE$_invasion_fleet_wandering_action_target }
					event_target:$TYPE$_invasion_fleet_wandering_action_target = {
						system_star = { prevprev = { auto_move_to_planet = { target = prev clear_auto_move_on_arrival = yes } } }
					}
				} 
				else_if = {
					limit = { any_system = {
						prev = { can_access_system = prev }
					} }
					random_system = {
						limit = { prev = { can_access_system = prev } }
						save_event_target_as = $TYPE$_invasion_fleet_wandering_action_target
					}
					event_target:$TYPE$_invasion_fleet_wandering_action_target = {
						system_star = { prevprev = { auto_move_to_planet = { target = prev clear_auto_move_on_arrival = yes } } }
					}
				}
				else = { 
					log = "跃迁强度：[$TYPE$_fleet_none_target_var]"
					change_variable = { which = $TYPE$_fleet_none_target_var value = 1 }
				}
			}
		}
	}

			NAND = {
				exists = orbit
				orbit = {
					OR = {
						AND = {
							exists = controller
							controller = { OR = { is_hostile = event_target:$OWNER$ is_hostile = event_target:$OWNER_2$ } }
							OR = { is_colony = yes is_under_colonization = yes }
						}
						AND = {
							always = $ATTACK_COLONIZABLE$
							is_colonizable = yes
						}
					}
				}
			}
		}
		clear_orders = yes
		clear_fleet_actions = this
		# 寻找所在星系目前没有己方单位的敌对星球
		if = {
			limit = {
				root = {
					exists = solar_system
					solar_system = {
						any_system_planet = { SRA_pathfinder_standard_system_target_triggers = { OWNER = $OWNER$ OWNER_2 = $OWNER_2$ ATTACK_COLONIZABLE = $ATTACK_COLONIZABLE$ } }
					}
				}
			}
			root = {
				solar_system = {
					random_system_planet = {
						limit = {
							SRA_pathfinder_standard_system_target_triggers = { OWNER = $OWNER$ OWNER_2 = $OWNER_2$ ATTACK_COLONIZABLE = $ATTACK_COLONIZABLE$ }
						}
						SRA_pathfinder_jump_drive_planet_action = yes
					}
				}
			}
		}

						SRA_pathfinder_standard_ship_target_triggers = { OWNER = $OWNER$ OWNER_2 = $OWNER_2$ }
					} }
				}

						limit = {
							SRA_pathfinder_standard_ship_target_triggers = { OWNER = $OWNER$ OWNER_2 = $OWNER_2$ }

						prevprev = { auto_follow_fleet = { target = prev.fleet attack_fleet = yes } }
