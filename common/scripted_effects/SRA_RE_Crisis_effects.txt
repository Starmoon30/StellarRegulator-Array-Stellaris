        }
        0 = {

        }
    }
}

#先驱舰队
create_SRA_RE_Crisis_1_CORE_fleet = {
	optimize_memory
    create_fleet = {
        name = "NAME_SRA_RE_Crisis_1_fleet"
        effect = {
            set_owner = event_target:SRA_RE_Crisis_country_1
            while = { 
                count = 45 
                create_ship = {
                    name = random
                    design = "NAME_SRA_RE_Core_Battleship"
                } 
            }
            set_location = { target = PREV distance = 5 angle = random }
            set_fleet_stance = aggressive
            set_fleet_bombardment_stance = SRA_doom2
            set_aggro_range_measure_from = self
            set_aggro_range = 500
            set_fleet_flag = SRA_RE_Crisis_1_fleet
            fleet_event = { id = SRA_RE_Crisis.39 }
            event_target:SRA_RE_Crisis_country_1 = { 
                change_variable = { which = "fleet_id" value = 1 }
            }
            set_variable = { which = "fleet_id" value = event_target:SRA_RE_Crisis_country_1.fleet_id }
            solar_system = { set_variable = { which = SRA_RE_Crisis_1_fleet_id value = prev.fleet_id } }
            }
        settings = { 
            can_upgrade = no 
            uses_naval_capacity = no
            can_change_leader = no
            can_change_composition = no
            can_disband = no
            spawn_debris = no
            ai_ignore_strength = yes
        }
    }
}
