initialize_SRA_Crisis_system_select = {
	optimize_memory
	every_system = {
		limit = { 
			OR = {
				any_country = {
					is_country_type = default
					capital_scope = {
						solar_system = {
							is_same_value = PREVPREVPREV
						}
					}
				}
			}
		}
		set_timed_star_flag = { flag = forbit_spawn_system days = 1 }
	}
	every_system = {
		limit = {
			NOT = { has_star_flag = forbit_spawn_system }
			OR = { #狡猾人老家		
				has_star_flag = Kuat_empire_system
				has_star_flag = kuat_initial_system
				has_star_flag = Kuat_system_int
				has_star_flag = Kuat_system_int_1
				has_star_flag = Kuat_system_int_2
				has_star_flag = Kuat_system_int_3
				has_star_flag = Kuat_system_int_4	
				has_star_flag = ag_kae_e_area_system 
				has_star_flag = starkiller_system
				has_star_flag = ag_no_jump_in_system
				has_star_flag = eternal_origin_starbase_megastructure_in_system
				closest_system = {
					max_steps = 3
					limit = {
						has_star_flag = forbit_spawn_system
					}
				}
			}
		}
		set_timed_star_flag = { flag = forbit_spawn_system days = 1 }
	}
}
