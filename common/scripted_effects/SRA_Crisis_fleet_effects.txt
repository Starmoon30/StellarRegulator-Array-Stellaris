create_SRA_Crisis_Cruiser_Fleet = {
	optimize_memory
    create_fleet = {
        name = "NAME_SRA_Crisis_Cruiser_Fleet"
        effect = {
            set_owner = event_target:SRA_Crisis_country			
            [[SRA_Crisis_Executor] set_variable = { which = num_SRA_Crisis_Executor value = $SRA_Crisis_Executor$ }
            set_variable = { which = num_SRA_Crisis_Executor_exist value = $SRA_Crisis_Executor$ }
            while = { count = $SRA_Crisis_Executor$ create_ship = { design = "NAME_SRA_Crisis_Executor" } }]

            [[SRA_Crisis_Cruiser] set_variable = { which = num_SRA_Crisis_Cruiser value = $SRA_Crisis_Cruiser$ }
            set_variable = { which = num_SRA_Crisis_Cruiser_exist value = $SRA_Crisis_Cruiser$ }
            while = { count = $SRA_Crisis_Cruiser$ create_ship = { design = "NAME_SRA_Crisis_Cruiser" } }]
            [[SRA_Crisis_Cruiser_No_Aura] set_variable = { which = num_SRA_Crisis_Cruiser_No_Aura value = $SRA_Crisis_Cruiser_No_Aura$ }
            set_variable = { which = num_SRA_Crisis_Cruiser_No_Aura_exist value = $SRA_Crisis_Cruiser_No_Aura$ }
            while = { count = $SRA_Crisis_Cruiser_No_Aura$ create_ship = { design = "NAME_SRA_Crisis_Cruiser_No_Aura" } }]
            set_location = { target = PREV distance = 5 angle = random }
            set_fleet_stance = aggressive
            set_fleet_bombardment_stance = SRA_doom2
            set_aggro_range_measure_from = self
            set_aggro_range = 500
			set_formation_scale = 5
            set_fleet_flag = SRA_attack_fleet
			set_timed_fleet_flag = { flag = SRA_Mutation_Blocked days = 1 }
            fleet_event = { id = SRA_Crisis.40 days = 5 }
            event_target:SRA_Crisis_country = { change_variable = { which = "fleet_id" value = 1 }}
            set_variable = { which = "fleet_id" value = event_target:SRA_Crisis_country.fleet_id }
            solar_system = { set_variable = { which = SRA_Crisis_fleet_id value = prev.fleet_id } }
            if = {
                limit = { has_global_flag = SRA_Mutation_2305 }
                fleet_event = {
                    id = SRA_Mutation.23051 days = 1
                }
            }
        }
        settings = { 
            can_upgrade = no 
            uses_naval_capacity = no
            can_change_leader = no
            can_change_composition = no
            can_disband = no
            spawn_debris = no
            ai_ignore_strength = yes
        }
    }
}
