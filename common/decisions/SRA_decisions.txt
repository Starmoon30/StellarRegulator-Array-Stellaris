	owned_planets_only = yes

	enactment_time = 3600
	resources = {
		category = decisions
		cost = {
			influence = 400
			minerals = 40000
			alloys = 20000
		}
	}

	potential = {
		owner = {
			OR = {
				has_civic = civic_SRAtechnology
				has_civic = civic_SRARociviaEmpire
		    }
		}
		has_modifier = planet_SRA_orbital_ring_Buff_1
		NOT = { SRA_is_SRA_planet = yes }
	}

	allow = {
		if = {
			limit = { owner = { is_ai = no } }
			custom_tooltip = {
				fail_text = legendary_leader_decision_arcology_project
				NOT = { has_planet_flag = legendary_leader_planet }
			}
		}
		num_uncleared_blockers < 1
	}

	effect = {
		# Make city-planet
		custom_tooltip = decision_SRA_city_project_effects
		hidden_effect = {
			clear_deposits = yes
			change_pc = pc_SRA_city
			owner = {
				country_event = { id = distar.201 } # port for living planet distar.197, triggers in event
			}
			while = {
				count = 5
				add_district = district_SRA_city_housing
			}
			carrier_event = { id = SRA_buildjobs.200 }
			set_planet_entity = {
				entity = "SRA_city_planet_01_entity"
			}
			if = {
				limit = {
					OR = {
						is_active_resolution = "resolution_ecology_environmental_control_board"
						is_active_resolution = "resolution_ecology_paradise_initiative"
					}
				}
				owner = {
					set_timed_country_flag = { flag = resolution_breached_terraformed days = 3600 }
				}
			}
		}
	}

	ai_weight = {
		weight = 100
	}
}

#星辉自动化
decision_SRA_auto_clear = {
	owned_planets_only = yes
	sound = event_administrative_work
	icon = decision_strategic_resources
	resources = {
		category = decisions
	}
	potential = {
		owner = {
			OR = {
				has_civic = civic_SRAtechnology
				has_civic = civic_SRARociviaEmpire
		    }
			has_country_flag = SRA_Automation_on
		}
	}
	effect = {
		#remove_all_districts = yes
		remove_all_buildings = yes
		set_owner = FROM
		if = {###旅游地
			limit = { has_modifier = resort_colony }
			add_building = building_resort_major_capital
		}
		else_if = {###流放地
			limit = { has_modifier = slave_colony }
			add_building = building_slave_major_capital
		}
		else_if = {###栖息地
			limit = { is_planet_class = pc_habitat }
			add_building = building_hab_system_capital
		}
		else_if = {###蜂群
			limit = { owner = { is_hive_empire = yes } }
			add_building = building_hive_major_capital
		}
		else_if = {###智械
			limit = { owner = { is_machine_empire = yes } }
			add_building = building_machine_system_capital
		}
		else = {###普通
			add_building = building_system_capital
		}
	}
	ai_weight = {
		weight = 0
	}
}
